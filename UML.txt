        +rom : ROM
        +screen : Screen
        +keyboard : Keyboard

        -timerService : TimerService

        +loadProgram(data: ByteArray)
        +start()
        +stop()
        +reset()
    }

package "CPU" {

    class CPU {
        +registers : Byte[8]
        +pc : Int
        +timer : Byte
        +address : Int
        +memoryStrategy : MemoryStrategy?
        +rom : MemoryDevice?
        +ram : RAM?
        +display : Display?
        +input : InputDevice?
        -terminated : Boolean

        +isHalted() : Boolean
        +cycle()
        +fetchInstruction() : Short
        +incrementPC()
        +skipInstruction()
        +switchMemory()
        +terminate(msg:String)
    }
}

package "Instructions" {

    abstract class Instruction <<Template Method>> {
        +execute(cpu:CPU)
        +decode(opcode:Short)
        #perform(cpu:CPU)
    }

    class StoreInstruction
    class HaltInstruction
    class AddInstruction
    class SubInstruction
    class ReadInstruction

    class WriteInstruction
    class JumpInstruction
    class ReadKeyboardInstruction
    class SwitchMemoryInstruction

    class SkipEqualInstruction
    class SkipNotEqualInstruction
    class SetAInstruction
    class SetTInstruction

    class ReadTInstruction
    class ConvertBase10Instruction
    class ConvertAsciiInstruction
    class DrawInstruction

    Instruction <|-- StoreInstruction
    Instruction <|-- AddInstruction
    Instruction <|-- SubInstruction
    Instruction <|-- ReadInstruction
    Instruction <|-- HaltInstruction
    Instruction <|-- WriteInstruction
    Instruction <|-- JumpInstruction
    Instruction <|-- ReadKeyboardInstruction
    Instruction <|-- SwitchMemoryInstruction
    Instruction <|-- SkipEqualInstruction
    Instruction <|-- SkipNotEqualInstruction
    Instruction <|-- SetAInstruction
    Instruction <|-- SetTInstruction
    Instruction <|-- ReadTInstruction
    Instruction <|-- ConvertBase10Instruction
    Instruction <|-- ConvertAsciiInstruction
    Instruction <|-- DrawInstruction

    StoreInstruction -[hidden]- AddInstruction
    AddInstruction -[hidden]- SubInstruction
    SubInstruction -[hidden]- ReadInstruction

    WriteInstruction -[hidden]- JumpInstruction
    JumpInstruction -[hidden]- ReadKeyboardInstruction
    ReadKeyboardInstruction -[hidden]- SwitchMemoryInstruction

    SkipEqualInstruction -[hidden]- SkipNotEqualInstruction
    SkipNotEqualInstruction -[hidden]- SetAInstruction
    SetAInstruction -[hidden]- SetTInstruction

    ReadTInstruction -[hidden]- ConvertBase10Instruction
    ConvertBase10Instruction -[hidden]- ConvertAsciiInstruction
    ConvertAsciiInstruction -[hidden]- DrawInstruction
}

package "Factory" {

    class InstructionFactory <<Factory>> {
        +create(opcode:Short) : Instruction
    }
}

package "Memory" {

    interface MemoryDevice {
        +read(address:Int) : Byte
        +write(address:Int,value:Byte)
    }

    class RAM {
        -data : Byte[4096]
    }

    class ROM {
        -data : Byte[4096]
        -writable : Boolean
    }

    MemoryDevice <|.. RAM
    MemoryDevice <|.. ROM
}

package "Strategy" {

    interface MemoryStrategy {
        +read(cpu:CPU) : Byte
        +write(cpu:CPU,value:Byte)
    }

    class RamStrategy
    class RomStrategy

    MemoryStrategy <|.. RamStrategy
    MemoryStrategy <|.. RomStrategy
}

package "IO" {

    interface Display {
        +draw(ascii:Byte,row:Int,col:Int)
        +render() : ByteArray
        +clear()
    }

    class Screen <<Singleton>> {
        -frameBuffer : Byte[64]

        +instance() : Screen
        +draw(ascii:Byte,row:Int,col:Int)
        +render() : ByteArray
        +clear()
    }

    interface InputDevice {
        +readHexByte() : Byte
    }

    class Keyboard {
        +readHexByte() : Byte
    }
}

package "Scheduling" {

    class CpuScheduler {
        +start()
        +stop()
    }

    class TimerService {
        +start()
        +stop()
        +tick()
    }
}
