package "Facade" {

    class D5700Computer <<Facade>> {
        +cpu : CPU
        +ram : RAM
        +rom : ROM
        +display : Display
        +keyboard : Keyboard
        -cpuScheduler : CpuScheduler
        -timerService : TimerService

        +loadProgram(data: ByteArray)
        +start()
        +stop()
        +reset()
    }

package "CPU" {

    class CPU {
        -registers : Byte[8]
        -pc : Int
        -timer : Byte
        -address : Int
        -memoryController : MemoryController
        -hardware : Hardware
        -terminated : Boolean

        +readRegister(index:Int) : Byte
        +writeRegister(index:Int,value:Byte)
        +jumpTo(address:Int)
        +incrementPC()
        +skipInstruction()
        +loadAddress(value:Int)
        +readMemory() : Byte
        +writeMemory(value:Byte)
        +decrementTimer()
        +readTimer() : Byte
        +drawRegister(registerIndex:Int,row:Int,column:Int)
        +readInputInto(registerIndex:Int)
        +cycle()
        +fetchInstruction() : Short
        +switchMemory()
        +terminate(msg:String)
    }

    class Hardware {
        -ram : MemoryDevice?
        -rom : MemoryDevice?
        -display : Display?
        -input : InputDevice?

        +getMemoryController() : MemoryController
        +getROM() : MemoryDevice?
        +draw(ascii:Byte,row:Int,column:Int)
        +readInput() : Byte
    }
}

package "Factory" {

    class InstructionFactory <<Factory>> {
        -creators : Map<Int, () -> Instruction>
        +create(opcode:Short) : Instruction
        -registerInstructions()
    }
}

package "Memory" {

    interface MemoryDevice {
        +read(address:Int) : Byte
        +write(address:Int,value:Byte)
    }

    class RAM {
        -data : Byte[4096]
    }

    class ROM {
        -data : Byte[4096]
        -writable : Boolean
    }

    class MemoryController {
        -ram : MemoryDevice?
        -rom : MemoryDevice?
        -strategy : MemoryStrategy?

        +read(cpu:CPU) : Byte
        +write(cpu:CPU,value:Byte)
        +switchMode()
        +reset()
        +hasMemory() : Boolean
    }

    MemoryDevice <|.. RAM
    MemoryDevice <|.. ROM
}

package "Strategy" {

    interface MemoryStrategy {
        +read(cpu:CPU) : Byte
        +write(cpu:CPU,value:Byte)
    }

    class RamStrategy
    class RomStrategy

    MemoryStrategy <|.. RamStrategy
    MemoryStrategy <|.. RomStrategy
}

package "IO" {

    interface Display {
        +draw(ascii:Byte,row:Int,col:Int)
        +render()
        +clear()
    }

    class Screen {
        -frameBuffer : Byte[64]

        +draw(ascii:Byte,row:Int,col:Int)
        +render()
        +clear()
    }

    Display <|.. Screen

    interface InputDevice {
        +readHexByte() : Byte
    }

    class Keyboard {
        +readHexByte() : Byte
    }

    InputDevice <|.. Keyboard
}

package "Scheduling" {

    class CpuScheduler {
        +start()
        +stop()
    }

    class TimerService {
        +start()
        +stop()
        +tick()
    }
}
